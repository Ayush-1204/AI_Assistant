#include<stdio.h>
#include<stdlib.h>

#define PAGE_COUNT 12

int pages[PAGE_COUNT] = {1, 3, 0, 3, 5, 6, 3, 0, 1, 2, 5, 3};

void fifo(int frame_size){
	int frames[frame_size], front=0, page_faults=0, found, i, j;

	for(i=0; i<frame_size; i++) frames[i]=-1;

	for(i=0; i<PAGE_COUNT; i++){
		found=0;
		for(j=0; j<frame_size; j++){
			if(frames[j]==pages[i]){
				found=1;
				break;
			}
		}
		if(!found){
			frames[front]=pages[i];
			front=(front+1)%frame_size;
			page_faults++;
		}
	}
	printf("\nFIFO Page Faults = %d\n", page_faults);
}

void lru(int frame_size){
	int frames[frame_size], recent[frame_size], time=0, page_faults=0, found, i, j, k, least_recent, index;

	for(i=0; i<frame_size; i++){
		frames[i]=-1;
		recent[i]=-1;
	}

	for(i=0; i<PAGE_COUNT; i++){
		found=0;
		for(j=0; j<frame_size; j++){
			if(frames[j]==pages[i]){
				found=1;
				recent[j]=time;
				break;
			}
		}
		if(!found){
			int empty=-1;
			for(j=0; j<frame_size; j++){
				if(frames[j]==-1){
					empty=j;
					break;
				}
			}
			if(empty!=-1){
				frames[empty]=pages[i];
				recent[empty]=time;
			}
			else{
				least_recent=recent[0];
				index=0;
				for(k=1; k<frame_size; k++){
					if(recent[k]<least_recent){
						least_recent=recent[k];
						index=k;
					}
				}
				frames[index]=pages[i];
				recent[index]=time;
			}
			page_faults++;
		}
		time++;
	}
	printf("LRU Page Faults = %d\n", page_faults);
}

int main(){
	int frame_size;

	printf("Enter frame size (minimum 3): ");
	scanf("%d", &frame_size);

	if(frame_size<3){
		printf("Frame size too small!\n");
		return 1;
	}

	printf("\nReference String: ");
	for(int i=0; i<PAGE_COUNT; i++)
		printf("%d ", pages[i]);
	printf("\n");

	fifo(frame_size);
	lru(frame_size);

	return 0;
}
